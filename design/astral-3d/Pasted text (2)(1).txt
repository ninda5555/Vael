5. sw.js

const CACHE_NAME = 'unsent-astral-v1';
const PUBLIC_ASSETS = [
  './',
  './index.html',
  './manifest.webmanifest',
  './assets/icon.svg',
  './styles/app.css',
  './src/app.js',
  './src/data/zodiac.js',
  './src/engine/astronomy.js',
  './src/engine/storage.js',
  './src/three/elements3d.js'
];

self.addEventListener('install', (event) => {
  event.waitUntil(
    caches.open(CACHE_NAME).then((cache) => cache.addAll(PUBLIC_ASSETS))
  );
  self.skipWaiting();
});

self.addEventListener('activate', (event) => {
  event.waitUntil(
    caches.keys().then((keys) =>
      Promise.all(keys.filter((k) => k !== CACHE_NAME).map((k) => caches.delete(k)))
    )
  );
  self.clients.claim();
});

self.addEventListener('fetch', (event) => {
  if (event.request.method !== 'GET') return;
  const url = new URL(event.request.url);
  if (url.origin !== self.location.origin) return;
  event.respondWith(
    caches.match(event.request).then((cached) => cached || fetch(event.request))
  );
});

import { SIGNS } from '../data/zodiac.js';

const RAD = Math.PI / 180;
export const mod360 = (x) => ((x % 360) + 360) % 360;

export function toJulianDay(date) {
  return date.getTime() / 86400000 + 2440587.5;
}

export function getSolarTropicalLongitude(date) {
  const T = (toJulianDay(date) - 2451545.0) / 36525.0;
  const L0 = 280.46646 + 36000.76983 * T + 0.0003032 * T * T;
  const M = (357.52911 + 35999.05029 * T - 0.0001537 * T * T) * RAD;
  const C = (1.914602 - 0.004817 * T - 0.000014 * T * T) * Math.sin(M)
          + (0.019993 - 0.000101 * T) * Math.sin(2 * M)
          + 0.000289 * Math.sin(3 * M);
  const omega = (125.04 - 1934.136 * T) * RAD;
  return mod360(L0 + C - 0.00569 - 0.00478 * Math.sin(omega));
}

export function getLunarLongitude(date) {
  const T = (toJulianDay(date) - 2451545.0) / 36525.0;
  const Lp = 218.3164477 + 481267.88123421 * T - 0.0015786 * T * T;
  const D  = (297.8501921 + 445267.1114034 * T - 0.0018819 * T * T) * RAD;
  const M  = (357.5291092 + 35999.0502909 * T) * RAD;
  const Mp = (134.9633964 + 477198.8675055 * T + 0.0087414 * T * T) * RAD;
  const F  = (93.2720950  + 483202.0175233 * T) * RAD;

  const sigmaL =
      6.288774 * Math.sin(Mp) + 1.274027 * Math.sin(2 * D - Mp) + 0.658314 * Math.sin(2 * D)
    + 0.213618 * Math.sin(2 * Mp) - 0.185116 * Math.sin(M) - 0.114332 * Math.sin(2 * F)
    + 0.058793 * Math.sin(2 * D - 2 * Mp) + 0.057066 * Math.sin(2 * D - M - Mp)
    + 0.053322 * Math.sin(2 * D + Mp) + 0.045758 * Math.sin(2 * D - M)
    - 0.040923 * Math.sin(M - Mp) - 0.034720 * Math.sin(D) - 0.030383 * Math.sin(M + Mp)
    + 0.015327 * Math.sin(2 * D - 2 * F) - 0.012528 * Math.sin(Mp + 2 * F) + 0.010980 * Math.sin(Mp - 2 * F);
  return mod360(Lp + sigmaL);
}

export function getLahiriAyanamsa(date) {
  const T = (toJulianDay(date) - 2451545.0) / 36525.0;
  return 23.853056 + 1.396971 * T;
}

export function getMoonPhaseInfo(date) {
  const angle = mod360(getLunarLongitude(date) - getSolarTropicalLongitude(date));
  const illumination = Math.round(((1 - Math.cos(angle * RAD)) / 2) * 100);
  let phaseName = 'Waxing Crescent';
  if (angle < 11.25 || angle >= 348.75) phaseName = 'New Moon';
  else if (angle < 78.75) phaseName = 'Waxing Crescent';
  else if (angle < 101.25) phaseName = 'First Quarter';
  else if (angle < 168.75) phaseName = 'Waxing Gibbous';
  else if (angle < 191.25) phaseName = 'Full Moon';
  else if (angle < 258.75) phaseName = 'Waning Gibbous';
  else if (angle < 281.25) phaseName = 'Last Quarter';
  else phaseName = 'Waning Crescent';
  return { angle: +angle.toFixed(2), illumination, phaseName };
}

export function findNextMoonEvent(startDate, targetAngle) {
  const stepMs = 6 * 3600 * 1000;
  let t1 = startDate.getTime();
  const signedDiff = (t) => {
    const a = mod360(getLunarLongitude(new Date(t)) - getSolarTropicalLongitude(new Date(t)));
    let d = a - targetAngle;
    if (d > 180) d -= 360;
    if (d < -180) d += 360;
    return d;
  };
  let d1 = signedDiff(t1);
  for (let i = 0; i < 140; i++) {
    const t2 = t1 + stepMs;
    const d2 = signedDiff(t2);
    if (d1 <= 0 && d2 >= 0 && Math.abs(d2 - d1) < 90) {
      let lo = t1, hi = t2;
      for (let b = 0; b < 18; b++) {
        const mid = (lo + hi) / 2;
        if (signedDiff(mid) < 0) lo = mid; else hi = mid;
      }
      return new Date((lo + hi) / 2);
    }
    t1 = t2; d1 = d2;
  }
  return null;
}

export function openIDB() {
  return new Promise((resolve) => {
    if (!window.indexedDB) { resolve(null); return; }
    const req = indexedDB.open('unsent_astral_vault', 1);
    req.onupgradeneeded = () => {
      const d = req.result;
      if (!d.objectStoreNames.contains('vault')) d.createObjectStore('vault', { keyPath: 'id' });
    };
    req.onsuccess = () => resolve(req.result);
    req.onerror = () => resolve(null);
  });
}

async function deriveAesKey(passphrase, salt) {
  const enc = new TextEncoder();
  const baseKey = await crypto.subtle.importKey('raw', enc.encode(passphrase), 'PBKDF2', false, ['deriveKey']);
  return crypto.subtle.deriveKey(
    { name: 'PBKDF2', salt, iterations: 120000, hash: 'SHA-256' },
    baseKey,
    { name: 'AES-GCM', length: 256 },
    false,
    ['encrypt', 'decrypt']
  );
}

export async function encryptVaultBackup(payloadObj, passphrase) {
  const salt = crypto.getRandomValues(new Uint8Array(16));
  const iv = crypto.getRandomValues(new Uint8Array(12));
  const key = await deriveAesKey(passphrase, salt);
  const encoded = new TextEncoder().encode(JSON.stringify(payloadObj));
  const cipher = await crypto.subtle.encrypt(
    { name: 'AES-GCM', iv, additionalData: new TextEncoder().encode('unsent-vault-v1') },
    key,
    encoded
  );
  return {
    version: 1,
    alg: 'AES-256-GCM',
    salt: btoa(String.fromCharCode(...salt)),
    iv: btoa(String.fromCharCode(...iv)),
    ciphertext: btoa(String.fromCharCode(...new Uint8Array(cipher)))
  };
}

function rebuildReleaseParticles() {
  const m = threeManagers.release;
  if (!m) return;
  while (m.holder.children.length) m.holder.remove(m.holder.children[0]);

  if (state.releaseMode === 'birds') {
    for (let i = 0; i < 16; i++) {
      const bird = new THREE.Group();
      const wingGeo = new THREE.BufferGeometry();
      wingGeo.setAttribute('position', new THREE.BufferAttribute(new Float32Array([0, 0, 0.12, 0.22, 0.05, 0, 0, 0, -0.14]), 3));
      const mat = new THREE.MeshBasicMaterial({ color: 0xF5EAD6, side: THREE.DoubleSide });
      const leftW = new THREE.Mesh(wingGeo, mat), rightW = new THREE.Mesh(wingGeo, mat);
      rightW.scale.x = -1; bird.add(leftW, rightW);
      bird.position.set((Math.random() - 0.5) * 2.8, -1.2 + Math.random() * 2.4, (Math.random() - 0.5) * 1.5);
      bird.userData = { leftW, rightW, phase: Math.random() * Math.PI * 2, speed: 0.012 + Math.random() * 0.01 };
      m.holder.add(bird);
    }
  } else {
    const isEmber = state.releaseMode === 'embers', n = 220;
    const geo = new THREE.BufferGeometry(), pos = new Float32Array(n * 3);
    for (let i = 0; i < n; i++) {
      pos[i * 3] = (Math.random() - 0.5) * 3.8; pos[i * 3 + 1] = (Math.random() - 0.5) * 2.4; pos[i * 3 + 2] = (Math.random() - 0.5) * 1.6;
    }
    geo.setAttribute('position', new THREE.BufferAttribute(pos, 3));
    const pts = new THREE.Points(geo, new THREE.PointsMaterial({ color: isEmber ? 0xFF8C42 : 0xB8B5C4, size: isEmber ? 0.065 : 0.05, transparent: true, opacity: 0.85 }));
    pts.userData.isEmber = isEmber; m.holder.add(pts);
  }
}

function loop3D() {
  requestAnimationFrame(loop3D);
  if (!state.reduceMotion) animT += 0.014;

  if (threeManagers.today && document.getElementById('s-today').classList.contains('active')) {
    const m = threeManagers.today, d = m.getDrag();
    m.elemHolder.children.forEach((c) => c.userData.update?.(animT));
    m.elemHolder.rotation.y = d.y; m.elemHolder.rotation.x = d.x * 0.4;
    if (!state.reduceMotion) m.ring.rotation.z = animT * 0.16;
    m.renderer.render(m.scene, m.camera);
  }
  if (document.getElementById('s-zodiac').classList.contains('active')) {
    if (state.zodiacView === 'elements' && threeManagers.element) {
      threeManagers.element.holder.children.forEach((c) => c.userData.update?.(animT));
      threeManagers.element.renderer.render(threeManagers.element.scene, threeManagers.element.camera);
    }
    if (state.zodiacView === 'wheel' && threeManagers.wheel) {
      const m = threeManagers.wheel;
      m.wheel.rotation.y += (m.targetRot - m.wheel.rotation.y) * (state.reduceMotion ? 1 : 0.09);
      m.renderer.render(m.scene, m.camera);
    }
    if (state.zodiacView === 'dialogue' && threeManagers.alchemy) {
      const m = threeManagers.alchemy;
      m.holderA.children.forEach((c) => c.userData.update?.(animT));
      m.holderB.children.forEach((c) => c.userData.update?.(animT + 1.2));
      m.renderer.render(m.scene, m.camera);
    }
  }
  if (threeManagers.card && document.getElementById('s-studio').classList.contains('active') && state.studioTab === 'card') {
    threeManagers.card.holder.children.forEach((c) => c.userData.update?.(animT));
    threeManagers.card.renderer.render(threeManagers.card.scene, threeManagers.card.camera);
  }
  if (threeManagers.release && document.getElementById('sh-release').classList.contains('open')) {
    const m = threeManagers.release;
    if (!state.reduceMotion) {
      m.holder.children.forEach((obj) => {
        if (obj.userData.leftW) {
          const flap = Math.sin(animT * 6 + obj.userData.phase) * 0.45;
          obj.userData.leftW.rotation.z = flap; obj.userData.rightW.rotation.z = -flap;
          obj.position.y += obj.userData.speed;
          if (obj.position.y > 1.35) obj.position.y = -1.25;
        } else if (obj.geometry?.attributes?.position) {
          const pos = obj.geometry.attributes.position, dir = obj.userData.isEmber ? 0.014 : -0.009;
          for (let i = 0; i < pos.count; i++) {
            let y = pos.getY(i) + dir;
            if (dir > 0 && y > 1.25) y = -1.25;
            if (dir < 0 && y < -1.25) y = 1.25;
            pos.setY(i, y);
          }
          pos.needsUpdate = true;
        }
      });
    }
    m.renderer.render(m.scene, m.camera);
  }
  if (threeManagers.desktop && window.innerWidth >= 980) {
    threeManagers.desktop.holder.children.forEach((c) => c.userData.update?.(animT));
    threeManagers.desktop.renderer.render(threeManagers.desktop.scene, threeManagers.desktop.camera);
  }
}

function resize3D() {
  Object.values(threeManagers).forEach((m) => {
    if (!m?.renderer) return;
    const wrap = m.renderer.domElement.parentElement;
    if (!wrap || !wrap.clientWidth || !wrap.clientHeight) return;
    m.camera.aspect = wrap.clientWidth / wrap.clientHeight;
    m.camera.updateProjectionMatrix();
    m.renderer.setSize(wrap.clientWidth, wrap.clientHeight, false);
  });
}

window.calculateBirthLens = function() {
  const dateVal = document.getElementById('birth-date-in').value;
  if (!dateVal) { showToast('Enter a birth date first'); return; }
  const unknown = document.getElementById('chk-unknown-time').checked;
  const timeVal = document.getElementById('birth-time-in').value || '12:00';
  const [, , tz] = document.getElementById('birth-city-in').value.split(',');
  const [Y, M, D] = dateVal.split('-').map(Number);
  const sampleDate = new Date(Date.UTC(Y, M - 1, D, 12, 0, 0));
  const offMin = Math.round((new Date(sampleDate.toLocaleString('en-US', { timeZone: tz })) - new Date(sampleDate.toLocaleString('en-US', { timeZone: 'UTC' }))) / 60000);
  const resBox = document.getElementById('lens-result-box'); resBox.classList.remove('hidden');

  if (unknown) {
    const sunSet = new Set(), moonSet = new Set();
    for (let h = 0; h <= 24; h += 1) {
      const d = new Date(Date.UTC(Y, M - 1, D, h, 0, 0) - offMin * 60000);
      sunSet.add(signFromLongitude(getSolarTropicalLongitude(d)).id);
      moonSet.add(signFromLongitude(mod360(getLunarLongitude(d) - getLahiriAyanamsa(d))).id);
    }
    const sunIds = [...sunSet], moonIds = [...moonSet];
    const sunLbl = sunIds.length === 1 ? `Birth Sun · ${getSign(sunIds[0]).western}` : `Birth Sun · ${getSign(sunIds[0]).western} or ${getSign(sunIds[1]).western} (time matters)`;
    const moonLbl = moonIds.length === 1 ? `Birth Moon (Lahiri) · ${getSign(moonIds[0]).rashi}` : `Birth Moon (Lahiri) · ${getSign(moonIds[0]).rashi} or ${getSign(moonIds[1]).rashi} (awaiting birth time)`;
    resBox.innerHTML = `<div class="label-ui" style="color:var(--text-primary);margin-bottom:4px;">${sunLbl}</div><div class="label-ui" style="color:var(--candle);margin-bottom:8px;">${moonLbl}</div><div style="display:flex;gap:8px;flex-wrap:wrap;"><button class="chip-btn" onclick="applyCalculatedLens('${sunIds[0]}', 'Birth Sun')">Use Sun lens</button>${moonIds.length === 1 ? `<button class="chip-btn" onclick="applyCalculatedLens('${moonIds[0]}', 'Birth Moon (Lahiri)')">Use Moon Rashi</button>` : ''}</div>`;
  } else {
    const [hh, mm] = timeVal.split(':').map(Number);
    const instant = new Date(Date.UTC(Y, M - 1, D, hh, mm, 0) - offMin * 60000);
    const sunDeg = getSolarTropicalLongitude(instant), ayanamsa = getLahiriAyanamsa(instant);
    const moonSidDeg = mod360(getLunarLongitude(instant) - ayanamsa);
    const sunSign = signFromLongitude(sunDeg), moonSign = signFromLongitude(moonSidDeg);
    resBox.innerHTML = `<div class="label-ui" style="color:var(--text-primary);">Birth Sun · ${sunSign.western} (${(sunDeg % 30).toFixed(1)}°)</div><div class="label-ui" style="color:var(--candle);margin:4px 0 8px;">Birth Moon (Lahiri) · ${moonSign.rashi} / ${moonSign.western} (${(moonSidDeg % 30).toFixed(1)}°)</div><div style="display:flex;gap:8px;flex-wrap:wrap;"><button class="chip-btn" onclick="applyCalculatedLens('${sunSign.id}', 'Birth Sun')">Use Birth Sun</button><button class="chip-btn" onclick="applyCalculatedLens('${moonSign.id}', 'Birth Moon (Lahiri)')">Use Birth Moon</button></div>`;
  }
};

  mode: z.enum(['reflect', 'poem', 'sign_question']),

// Transactional daily quota guard (prevents uncontrolled spend)
async function reserveQuota(uid: string, limitPerDay = 20): Promise<void> {
  const day = new Date().toISOString().slice(0, 10);
  const ref = db.doc(`quotas/${uid}_${day}`);
  await db.runTransaction(async (tx) => {
    const snap = await tx.get(ref);
    const used = snap.exists ? (snap.data()?.count ?? 0) : 0;
    if (used >= limitPerDay) throw new HttpsError('resource-exhausted', 'Daily limit reached.');
    tx.set(ref, { count: used + 1, updatedAt: admin.firestore.FieldValue.serverTimestamp() }, { merge: true });
  });
}

// Non-computational Gemini reflection endpoint (Enforce App Check in production)
export const generateQuietReflection = onCall(

    if (!request.auth) throw new HttpsError('unauthenticated', 'Sign in required for cloud reflection.');
